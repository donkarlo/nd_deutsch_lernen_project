\section{Kurze Grundregel}

\begin{grammarentry}{gegen}{Präposition + Akkusativ}
\GermanText{\glqq gegen\grqq ist als Präposition immer mit dem Akkusativ verbunden.}
\EnglishText{As a preposition, \glqq gegen\grqq is always used with the accusative.}

\GermanText{Die wichtigsten Grundideen sind: Richtung zu etwas hin, Widerstand/Konflikt, ungefährer Zeitpunkt, Vergleich/Austausch und Schutz vor etwas.}
\EnglishText{The main basic ideas are: direction toward something, opposition/conflict, approximate time, comparison/exchange, and protection against something.}
\end{grammarentry}

\begin{exbox}
\begin{itemize}
    \ExampleItem{Er läuft gegen die Wand.}{He runs into the wall.}
    \ExampleItem{Wir sind gegen diese Entscheidung.}{We are against this decision.}
    \ExampleItem{Ich komme gegen acht Uhr.}{I will come around eight o'clock.}
    \ExampleItem{Dieses Medikament hilft gegen Schmerzen.}{This medicine helps against pain.}
\end{itemize}
\end{exbox}

\EnglishText{After \glqq gegen\grqq, the accusative is used.}

\section{Bedeutung 1: Richtung und Kontakt}

\begin{grammarentry}{gegen + Akkusativ}{against / into}
\GermanText{\glqq gegen\grqq kann eine Bewegung in Richtung einer Sache zeigen, oft mit Kontakt oder Zusammenstoß.}
\EnglishText{\glqq gegen\grqq can show movement toward something, often with contact or collision.}
\end{grammarentry}

\begin{exbox}
\begin{itemize}
    \ExampleItem{Der Ball fliegt gegen die Tür.}{The ball flies against the door.}
    \ExampleItem{Ich stoße gegen den Tisch.}{I bump into the table.}
    \ExampleItem{Das Auto fährt gegen einen Baum.}{The car drives into a tree.}
    \ExampleItem{Er lehnt sich gegen die Wand.}{He leans against the wall.}
\end{itemize}
\end{exbox}

\section{Bedeutung 2: Widerstand oder Meinung}

\EnglishText{\glqq gegen\grqq often means that one rejects something or has an opposing opinion.}

\end{exbox}

\section{Bedeutung 4: ungefährer Zeitpunkt}

\begin{grammarentry}{gegen + Uhrzeit}{around / approximately at}
\GermanText{Mit Uhrzeiten bedeutet \glqq gegen\grqq ungefähr. Es ist nicht so genau wie \glqq um\grqq.}
\EnglishText{With clock times, \glqq gegen\grqq means approximately. It is not as exact as \glqq um\grqq.}
\end{grammarentry}

\begin{exbox}
\begin{itemize}
    \ExampleItem{Ich komme gegen acht Uhr.}{I come around eight o'clock.}
    \ExampleItem{Der Zug kommt gegen Mittag an.}{The train arrives around noon.}
    \ExampleItem{Wir treffen uns gegen Abend.}{We meet toward evening / around evening.}
\end{itemize}
\end{exbox}

\section{Häufige Fehler}

\begin{grammarentry}{Fehler 1}{Dativ nach gegen}
\GermanText{Nach \glqq gegen\grqq benutzt man nicht den Dativ, sondern den Akkusativ.}
\EnglishText{After \glqq gegen\grqq, one does not use the dative, but the accusative.}
\begin{itemize}
    \item \Wrong{Ich bin gegen dem Plan.}
    \item \Correct{Ich bin gegen den Plan.}
\end{itemize}
\end{grammarentry}

\begin{grammarentry}{Fehler 2}{gegen und um bei Uhrzeit}
\GermanText{\glqq um acht Uhr\grqq ist normalerweise genau. \glqq gegen acht Uhr\grqq ist ungefähr.}
\EnglishText{\glqq um acht Uhr\grqq is normally exact. \glqq gegen acht Uhr\grqq is approximate.}
\begin{itemize}
    \item \Correct{Der Termin beginnt um acht Uhr.} \hfill exact
    \item \Correct{Ich komme gegen acht Uhr.} \hfill approximate
\end{itemize}
\end{grammarentry}

\begin{grammarentry}{Fehler 3}{helfen mit Dativ, aber gegen mit Akkusativ}
\GermanText{Das Verb \glqq helfen\grqq braucht Dativ für die Person. Die Präposition \glqq gegen\grqq braucht Akkusativ für das Problem.}
\EnglishText{The verb \glqq helfen\grqq needs dative for the person. The preposition \glqq gegen\grqq needs accusative for the problem.}
\begin{itemize}
    \item \Correct{Der Arzt hilft mir.}
    \item \Correct{Das Medikament hilft gegen die Schmerzen.}
    \item \Correct{Der Arzt hilft mir gegen die Schmerzen.}
\end{itemize}
\end{grammarentry}

\section{Kurze Übung}

\GermanText{Ergänze die richtige Form nach \glqq gegen\grqq.}
\EnglishText{Complete the correct form after \glqq gegen\grqq.}

\begin{enumerate}
    \item Ich bin gegen \underline{\hspace{2.5cm}} Plan. \hfill der Plan
    \item Das Auto fährt gegen \underline{\hspace{2.5cm}} Wand. \hfill die Wand
    \item Das verstößt gegen \underline{\hspace{2.5cm}} Gesetz. \hfill das Gesetz
    \item Dieses Medikament hilft gegen \underline{\hspace{2.5cm}} Schmerzen. \hfill die Schmerzen
\end{enumerate}

\subsection{Lösung}

\begin{enumerate}
    \item Ich bin gegen \Correct{den} Plan.
    \item Das Auto fährt gegen \Correct{die} Wand.
    \item Das verstößt gegen \Correct{das} Gesetz.
    \item Dieses Medikament hilft gegen \Correct{die} Schmerzen.
\end{enumerate}
