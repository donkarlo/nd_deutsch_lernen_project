\section{Bedeutung und Gesprächsfunktion \textnormal{(Meaning and pragmatic function)}}
\begin{grammarentry}{nur}{Modalpartikel / Abtönungspartikel}
\GermanText{Nur verstärkt als Modalpartikel emotionale Fragen, Warnungen oder Wünsche.}
\EnglishText{As a modal particle, nur intensifies emotional questions, warnings and wishes.}
\end{grammarentry}
\section{Satzart, Stellung und Betonung \textnormal{(Sentence mood, position and stress)}}
\begin{grammarentry}{Verwendung}{Modalpartikel / Abtönungspartikel}
\GermanText{Besonders häufig in W-Fragen und Wunschsätzen, gelegentlich in warnenden Aufforderungen.}
\EnglishText{Especially common in wh-questions and wishes, sometimes in warnings.}
\end{grammarentry}
\GermanText{Als Abtönungspartikel ist nur unflektierbar und verlangt keinen Kasus. Sie steht normalerweise im Mittelfeld und kann das Vorfeld eines normalen Aussagesatzes nicht allein besetzen.}
\EnglishText{As a discourse particle, nur is invariable and governs no case. It normally occurs in the middle field and cannot stand alone in the prefield of a declarative sentence.}
\section{Typische Beispiele \textnormal{(Typical examples)}}
\begin{exbox}\begin{itemize}
\ExampleItem{Wie soll ich das nur schaffen?}{How on earth am I supposed to manage?}
\ExampleItem{Wenn ich nur mehr Zeit hätte!}{If only I had more time!}
\ExampleItem{Wo ist nur mein Schlüssel?}{Where on earth is my key?}
\ExampleItem{Pass nur auf!}{You'd better watch out!}
\end{itemize}\end{exbox}
\section{Andere grammatische Rollen \textnormal{(Other grammatical roles)}}
\begin{grammarentry}{Abgrenzung}{Modalpartikel / Abtönungspartikel}
\GermanText{Als Fokuspartikel bedeutet nur lediglich oder ausschließlich und begrenzt eine Zahl oder Gruppe.}
\EnglishText{As a focus particle, nur means only or exclusively, restricting a number or set.}
\end{grammarentry}
\begin{exbox}\begin{itemize}
\ExampleItem{Ich habe nur zwei Euro.}{I have only two euros.}
\ExampleItem{Was soll ich nur tun?}{What on earth should I do?}
\end{itemize}\end{exbox}
\section{Lernhinweis \textnormal{(Learning tip)}}
\GermanText{Nicht die Wortform allein, sondern der Kontext entscheidet, ob nur als Modalpartikel verwendet wird. Die englische Übersetzung kann die kommunikative Absicht statt einer wortwörtlichen Entsprechung ausdrücken.}
\EnglishText{The word form alone does not determine whether nur is a modal particle. English translation may express its conversational intent instead of using a word-for-word equivalent.}
