\section{Grundbedeutung und grammatische Rolle}

\begin{grammarentry}{wer}{W-Relativpronomen, Nominativ}

\GermanText{\textbf{wer} kann einen \textbf{freien Relativsatz} einleiten. Es bedeutet ungefähr \glqq whoever; anyone who\grqq{}.}

\EnglishText{\textbf{wer} can introduce a \textbf{free relative clause}. It means approximately \glqq whoever; anyone who\grqq{}.}

\GermanText{Im Relativsatz hat \textbf{wer} die Funktion \textbf{Subjekt}.}

\EnglishText{Inside the relative clause, \textbf{wer} functions as \textbf{Subjekt}.}

\GermanText{\textbf{wer} ist die Nominativform der Reihe \textbf{wer - wen - wem - wessen}.}

\EnglishText{\textbf{wer} is the nominativ form in the series \textbf{wer - wen - wem - wessen}.}

\GermanText{Der Kasus wird durch die Funktion von \textbf{wer} im Relativsatz bestimmt, nicht durch die Funktion des ganzen Relativsatzes im Hauptsatz.}

\EnglishText{The case is determined by the function of \textbf{wer} inside the relative clause, not by the function of the whole relative clause in the main clause.}

\end{grammarentry}

\section{Freier Relativsatz}

\begin{grammarentry}{Kein ausdrückliches Bezugsnomen}{Bedeutung bereits enthalten}

\GermanText{Vor \textbf{wer} steht normalerweise kein konkretes Bezugsnomen. Die Bedeutung ist ungefähr \glqq die Person, die ...\grqq{} oder \glqq jeder, der ...\grqq{}.}

\EnglishText{There is normally no concrete antecedent noun before \textbf{wer}. The meaning is approximately \glqq the person who ...\grqq{} or \glqq whoever ...\grqq{}.}

\GermanText{Wenn bereits ein Nomen vorhanden ist, nimmt man normalerweise \textbf{der/die/das} und die passenden deklinierten Formen.}

\EnglishText{If a noun is already present, German normally uses \textbf{der/die/das} and the appropriate declined forms.}

\end{grammarentry}

\section{Beispiele}

\begin{exbox}

\begin{itemize}

\ExampleItem{Wer viel übt, lernt schneller.}{Whoever practices a lot learns faster.}

\ExampleItem{Wer mir hilft, dem helfe ich auch.}{Whoever helps me, I help as well.}

\ExampleItem{Wer das sagt, irrt sich.}{Whoever says that is mistaken.}

\end{itemize}

\end{exbox}

\section{Typischer Fehler}

\begin{grammarentry}{Nicht mit einem normalen Bezugsnomen}{Korrektur}

\GermanText{Falsch: \textbf{Der Mann, wer dort steht, ist mein Lehrer.}}

\EnglishText{Incorrect: Der Mann, wer dort steht, ist mein Lehrer.}

\GermanText{Richtig: \textbf{Der Mann, der dort steht, ist mein Lehrer.}}

\EnglishText{Correct: Der Mann, der dort steht, ist mein Lehrer.}

\end{grammarentry}

\section{Wortstellung und Komma}

\begin{grammarentry}{Nebensatzregel}{Verb am Ende}

\GermanText{Der freie Relativsatz ist ein Nebensatz. Das konjugierte Verb steht normalerweise am Ende. Der Relativsatz wird durch Kommas abgetrennt.}

\EnglishText{A free relative clause is a subordinate clause. The conjugated verb normally stands at the end. The relative clause is separated by commas.}

\end{grammarentry}

\section{Merksatz}

\begin{grammarentry}{wer}{kurz und wichtig}

\GermanText{\textbf{wer} = Nominativform eines W-Relativpronomens für Personen in freien Relativsätzen.}

\EnglishText{\textbf{wer} = the nominativ form of a W-relative pronoun for people in free relative clauses.}

\end{grammarentry}
