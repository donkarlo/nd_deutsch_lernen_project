    \section{Schreiben (ÖIF) --- Answer Structure}
        
        \subsection{1. Salutation (Anrede)}
            \begin{itemize}
                \item Formal:
                \begin{itemize}
                    \item Sehr geehrte Damen und Herren,
                    \item Sehr geehrte Frau \dots\ / Sehr geehrter Herr \dots,
                \end{itemize}
                \item Informal:
                \begin{itemize}
                    \item Liebe \dots\ / Lieber \dots,
                \end{itemize}
            \end{itemize}
        
        \subsection{2. Opening (Einleitung)}
            Write 1--2 sentences stating your purpose clearly and directly.
            \begin{itemize}
                \item Ich schreibe Ihnen, weil \dots
                \item Ich moechte Sie ueber \dots\ informieren.
                \item Ich habe Ihre Anzeige gelesen und \dots
            \end{itemize}
        
        \subsection{3. Main part (Hauptteil)}
            Answer every point from the prompt (usually 3--4 items). Missing a point reduces the score.
            \begin{itemize}
                \item Use one short paragraph or 1--2 sentences per point.
                \item Keep sentences simple but correct (verb position matters).
                \item Prefer clarity over complex structures.
            \end{itemize}
        
        \subsection{4. Closing (Schluss)}
            \begin{itemize}
                \item Formal:
                \begin{itemize}
                    \item Ich freue mich auf Ihre Antwort.
                    \item Vielen Dank im Voraus.
                \end{itemize}
                \item Informal:
                \begin{itemize}
                    \item Ich freue mich, von dir zu hoeren.
                \end{itemize}
            \end{itemize}
        
        \subsection{5. Sign-off (Grussformel)}
            \begin{itemize}
                \item Formal: Mit freundlichen Grüssen
                \item Informal: Viele Gruesse / Liebe Gruesse
            \end{itemize}
        
        \subsection{What examiners typically check}
            \begin{itemize}
                \item Did you answer all prompt points?
                \item Is the text logically structured and easy to follow?
                \item Are basic grammar and word order correct (especially the verb position)?
                \item Are the required number of words met?
            \end{itemize}
        
        \subsection{Practical tip (B1 and below)}
            Simple, correct sentences score better than complex, incorrect ones.
            \begin{itemize}
                \item Instead of: Obwohl ich sehr beschaeftigt gewesen bin, habe ich versucht \dots
                \item Write: Ich war sehr beschaeftigt. Trotzdem habe ich versucht \dots
            \end{itemize}
