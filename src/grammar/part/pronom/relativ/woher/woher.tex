\section{Grundbedeutung und Kategorie}

\begin{grammarentry}{woher}{Relativadverb, Herkunft}

\GermanText{\textbf{woher} kann in relativer Funktion einen Ausgangsort oder eine Herkunft verbinden. Seine ungefähre Bedeutung ist \glqq where from; from which place\grqq{}.}

\EnglishText{\textbf{woher} can in relative use connect a place of origin or source. Its approximate meaning is \glqq where from; from which place\grqq{}.}

\end{grammarentry}

\section{Wichtig: Pronomen oder Adverb?}

\begin{grammarentry}{Grammatische Einordnung}{kein deklinierbares Pronomen}

\GermanText{\textbf{woher} ist in dieser Verwendung kein deklinierbares Relativpronomen. Es ist ein Relativadverb bzw. relatives Verbindungswort. Deshalb hat es keinen Nominativ, Akkusativ, Dativ oder Genitiv.}

\EnglishText{\textbf{woher} is not a declinable relative pronoun in this use. It is a relative adverb or relative linking word. Therefore, it has no nominative, accusative, dative, or genitive case.}

\GermanText{In formeller Sprache sind bei einem konkreten Nomen Formen wie \textbf{aus dem / aus der / von dem} oft natürlicher.}

\EnglishText{In formal language, forms such as \textbf{aus dem / aus der / von dem} are often more natural with a concrete noun.}

\end{grammarentry}

\section{Beispiele}

\begin{exbox}

\begin{itemize}

\ExampleItem{Das ist die Stadt, woher er kommt.}{That is the city he comes from.}

\ExampleItem{Sie zeigte mir den Ort, woher das Geräusch kam.}{She showed me the place the noise came from.}

\end{itemize}

\end{exbox}

\section{Wortstellung und Komma}

\begin{grammarentry}{Nebensatzregel}{Verb am Ende}

\GermanText{Der eingeleitete Satz ist ein Nebensatz. Das konjugierte Verb steht normalerweise am Ende, und der Satz wird durch Kommas abgetrennt.}

\EnglishText{The introduced clause is a subordinate clause. The conjugated verb normally stands at the end, and the clause is separated by commas.}

\end{grammarentry}

\section{Merksatz}

\begin{grammarentry}{woher}{kurz und wichtig}

\GermanText{\textbf{woher} kann relativ verwendet werden, ist aber grammatisch ein \textbf{Relativadverb/Relativwort}, kein Relativpronomen wie \textbf{wer} oder \textbf{was}.}

\EnglishText{\textbf{woher} can be used relatively, but grammatically it is a \textbf{relative adverb/relative word}, not a relative pronoun like \textbf{wer} or \textbf{was}.}

\end{grammarentry}
