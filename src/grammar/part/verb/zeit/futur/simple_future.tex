\section*{Futur I}
    
    \textbf{Form:} werden (conjugated) + infinitive (at the end)
    
    \begin{itemize}
        \item ich werde machen
        \item du wirst machen
        \item er/sie/es wird machen
        \item wir werden machen
        \item ihr werdet machen
        \item sie werden machen
    \end{itemize}
    
    \textbf{Beispiel mit \emph{haben}:}
    
    \begin{itemize}
        \item ich werde haben
        \item du wirst haben
        \item er/sie/es wird haben
    \end{itemize}
